
                        function startExams(confirm = 0, agreed_not_save = 0) {
                            if (agreed_not_save == 1) {
                                $("#alert-not-in-site").modal('hide');
                            }
                            $.ajax({
                                type: 'POST',
                                url: nv_base_siteurl + 'index.php?' + nv_lang_variable + '=' + nv_lang_data +
                                    '&' + nv_name_variable + '=' + nv_module_name + '&' + nv_fc_variable +
                                    '=detail&nocache=' + new Date().getTime(),
                                data : 'sendinfo=1&id=523&code=' + $('#code').val() + '&price=0' + exam_subject_id + '&confirm=' + confirm + '&agreed_not_save=' + agreed_not_save,
                                success: function(data) {
                                    var r_split = data.split('_');
                                    if (r_split[0] == 'OK') {
                                        modalShow('Thông báo!', '<p>' + r_split[1] + '</p>' + '<div class="text-center"><div class="load-bar"></div></div>');
                                        setTimeout(function(){
                                            nv_test_unset_answer();
                                            nv_test_unset_limit_audio();
                                            window.location.href = window.location.href;
                                        }, 1500)
                                    }
                                    else if (r_split[0] == 'NOTCONFIRM') {
                                        $("#alert-before-exam").modal();
                                    }
                                    else if (r_split[0] == 'NOTENOUGHMONEY') {
                                        $("#remain_money").text(r_split[1]);
                                        $("#alert-not-enough-money").modal();
                                    }
                                    else if (r_split[0] == 'NOTINSITE') {
                                        $("#alert-not-in-site").modal();
                                    }
                                    else if (r_split[0] == 'QUIZGAME') {
                                        if (r_split[1] !== undefined && r_split[1] !== '') {
                                            $('#quiz-game-info-guest input[name="guest_name"]').val(JSON.parse(r_split[1]));
                                        }
                                        $("#quiz-game-info-guest").modal();
                                    }
                                    else {
                                        alert(r_split[1]);
                                    }
                                }
                            });
                        }
                        $(document).ready(function() {
                            $('#quiz-game-info-guest button[type="submit"]').click(function() {
                                let quiz_game_info_guest = $('#quiz-game-info-guest form').serializeArray();
                                // Kiểm tra dữ liệu
                                let check = true;
                                for (let item of quiz_game_info_guest) {
                                    check = check && (trim(item.value) !== '');
                                }
                                if (!check) {
                                    alert('Vui lòng điền đầy đủ thông tin');
                                    return !1;
                                }
                                $('#quiz-game-info-guest button[type="submit"]').prop('disabled', true);
                                $.ajax({
                                    type: 'POST',
                                    url: nv_base_siteurl + 'index.php?' + nv_lang_variable + '=' + nv_lang_data +
                                        '&' + nv_name_variable + '=' + nv_module_name + '&' + nv_fc_variable +
                                        '=detail&quiz_game_info_guest=1&id=523&nocache=' + new Date().getTime(),
                                    data : $('#quiz-game-info-guest form').serialize(),
                                    success: function(data) {
                                        var r_split = data.split('_');
                                        if (r_split[0] == 'OK') {
                                            startExams()
                                        }                                       
                                        else {
                                            alert(r_split[1]);
                                        }
                                    }
                                });
                            });
                            $('#btn-save').click(function() {
                                $.ajax({
                                    type: 'POST',
                                    dataType: "json",
                                    url: nv_base_siteurl + 'index.php?' + nv_lang_variable +
                                        '=' + nv_lang_data + '&' + nv_name_variable + '=' +
                                        nv_module_name + '&' + nv_fc_variable +
                                        '=detail&nocache=' + new Date().getTime(),
                                    data : 'save_exam=1&id=523',
                                    success: function(data) {
                                        if (data.status == 0) {
                                            alert(data.message);
                                            loginForm('');
                                        } else if (data.status == 2) {
                                            modalShow('', data.message);
                                        } else if (data.status == 1) {
                                            $('#btn-save').attr('disabled', true);
                                            $('#btn-save').html('<i class="fa fa-check" aria-hidden="true"></i>&nbsp;Đã lưu đề thi');
                                        }
                                    }
                                });
                            });
                        });
